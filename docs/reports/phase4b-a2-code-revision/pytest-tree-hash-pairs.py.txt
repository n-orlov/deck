"""Task 005 (approach 2) — pytest verification of the pinned code revision.

This file is NOT part of the deck repo's test surface: the repo's tests are Go
test functions and godog scenarios only, and no test framework may be added to
it. It is a record-verification harness for task 005's own claims, kept in the
run's artifacts directory and committed into the task's report directory as a
plain listing (``pytest-tree-hash-pairs.py.txt``) so the checks can be re-run
by hand at any later commit without ever being collected by the repo's tests.

Run it as::

    DECK_REPO=/workspace python3 -m pytest -q <this file>

Optional ``DECK_TASK_COMMIT`` (default ``HEAD``) names the commit whose path
trees are compared against the pinned revision's path trees.
"""

import os
import subprocess

import pytest

REPO = os.environ.get("DECK_REPO", "/workspace")
TASK_COMMIT = os.environ.get("DECK_TASK_COMMIT", "HEAD")

# The pinned code revision: the last commit touching a .go or .feature file
# (finding R1's cure commit, task 004).
REV = "70c7430df3b23a46fb735e8573e26ec55908adeb"

# The four labelled tree-object hashes downstream tasks 006-012 cite.
EXPECTED_TREE_HASHES = {
    "internal": "15506734d4989e111e871a419ebf46c94a3b59a3",
    "cmd": "27ff2eba72ef6a63a6cf49285c4ddc6660b7fb0d",
    "features": "b5dbe2f565eb96b2654f8d1d64fcc51eb711f60d",
    "ci": "0a183631a2beea070ab0f7d8fa027aecf423e7b0",
}

# The four cure commits (tasks 001-004 / findings B1, B2, B3, R1) leading to REV.
CURE_COMMITS = {
    "001/B1": "113b552ca2fe42bb1b3d10e42f14cdc3f735c8e6",
    "002/B2": "075c59c35fbbc22461f748bbe87e87a8ac4139e4",
    "003/B3": "00f33a5ac204bc2f558352a18c40fefa24ad3bab",
    "004/R1": "70c7430df3b23a46fb735e8573e26ec55908adeb",
}


def git(*args):
    out = subprocess.run(
        ["git", "-C", REPO, *args],
        capture_output=True,
        text=True,
        check=True,
    )
    return out.stdout.strip()


def test_pinned_rev_is_the_last_code_touching_commit():
    """`git log -1 --format=%H -- '*.go' '*.feature'` at the task commit == REV."""
    assert git("log", "-1", "--format=%H", TASK_COMMIT, "--", "*.go", "*.feature") == REV


@pytest.mark.parametrize("path", sorted(EXPECTED_TREE_HASHES))
def test_tree_hash_pair_equal_at_rev_and_task_commit(path):
    """The pair (REV:<path>, <task commit>:<path>) is equal, and is the pinned hash.

    Equality of the pair is what shows the record-only commit did not move code.
    """
    at_rev = git("rev-parse", f"{REV}:{path}")
    at_task_commit = git("rev-parse", f"{TASK_COMMIT}:{path}")
    assert at_rev == at_task_commit, f"{path}: {at_rev} != {at_task_commit}"
    assert at_rev == EXPECTED_TREE_HASHES[path]


def test_record_only_commits_changed_no_go_or_feature_file():
    """Corroboration (not the identity proof): REV..task commit touches no code file."""
    changed = [
        line
        for line in git("diff", "--name-only", REV, TASK_COMMIT).splitlines()
        if line.endswith(".go") or line.endswith(".feature")
    ]
    assert changed == []


@pytest.mark.parametrize("label", sorted(CURE_COMMITS))
def test_cure_commit_resolves(label):
    """`git cat-file -e <sha>^{commit}` exits 0 for each of the four cure commits."""
    sha = CURE_COMMITS[label]
    rc = subprocess.run(
        ["git", "-C", REPO, "cat-file", "-e", f"{sha}^{{commit}}"],
        capture_output=True,
        text=True,
    ).returncode
    assert rc == 0, f"{label} {sha} does not resolve as a commit object (exit {rc})"
