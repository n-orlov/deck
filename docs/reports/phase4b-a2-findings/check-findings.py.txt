"""Task 010 (approach 2) -- pytest verification for docs/reports/phase4b-findings.md's
approach-2 addendum.

Not part of the deck repo's test surface: the repo's tests are Go test
functions and godog scenarios only, no other framework is introduced. This
file is committed as a plain listing (`.py.txt`) beside the findings it
checks, so it can be re-run by hand at any later commit without ever being
collected by the repo's own suite.

Run it as::

    DECK_REPO=/workspace python3 -m pytest -q <this file>

Two things it verifies, matching the two clauses of task 010's own
criterion that name a check:

1. The failure-list clause: task 008's README failure list is bounded by
   the ten committed stability-sweep logs, and re-running the same
   `grep -n 'FAIL'` those ten logs quantify over still produces empty
   output (i.e. `none`), so the findings addendum's "none, task 008's list
   is empty" claim is not stale.
2. Every `file:line` the findings addendum's approach-2 section adds
   resolves at DECK_READ_COMMIT (default: this file's own commit, once
   committed -- pass a different commit via DECK_READ_COMMIT to check an
   earlier state without a self-reference).
"""

import os
import subprocess

import pytest

REPO = os.environ.get("DECK_REPO", "/workspace")

# This file's own commit, once committed. A caller checking an EARLIER
# state (e.g. before this task's commit exists) must override this via
# DECK_READ_COMMIT -- the default is intentionally this task's own commit
# so a bare `pytest` run after committing checks the real, final tree.
READ_COMMIT = os.environ.get("DECK_READ_COMMIT", "HEAD")

STABILITY10_DIR = "docs/reports/phase4b-a2-stability10"
STABILITY10_LOGS = [f"run-{n}.log" for n in range(1, 11)]


def git(*args):
    out = subprocess.run(
        ["git", "-C", REPO, *args],
        capture_output=True,
        text=True,
        check=True,
    )
    return out.stdout


def git_line(rev, path, lineno):
    """Return the 1-indexed `lineno`'th line of `path` as it reads at `rev`."""
    content = git("show", f"{rev}:{path}")
    lines = content.splitlines()
    assert 1 <= lineno <= len(lines), (
        f"{path} at {rev} has only {len(lines)} lines, wanted line {lineno}"
    )
    return lines[lineno - 1]


def test_failure_list_is_still_empty_across_the_ten_committed_logs():
    """Task 008's own failure-grep re-run, repeated here: still empty."""
    log_paths = [os.path.join(REPO, STABILITY10_DIR, n) for n in STABILITY10_LOGS]
    for p in log_paths:
        assert os.path.isfile(p), f"missing committed log: {p}"
    proc = subprocess.run(
        ["grep", "-n", "FAIL", *log_paths],
        capture_output=True,
        text=True,
    )
    # grep exit 1 == no lines matched == still empty, matching task 008's
    # own committed failure-grep.log and the addendum's "none" claim.
    assert proc.returncode == 1, (
        f"expected no FAIL lines across the ten committed logs, got "
        f"exit {proc.returncode}, stdout={proc.stdout!r}"
    )
    assert proc.stdout == ""


def test_vg6_spec_line_is_the_plain_primary_key_form():
    line = git_line(READ_COMMIT, "SPEC.md", 332)
    assert "id" in line and "INTEGER PRIMARY KEY" in line and "AUTOINCREMENT" not in line, (
        f"SPEC.md:332 no longer reads as the plain form the finding cites: {line!r}"
    )


def test_vg6_code_line_is_the_autoincrement_form():
    line = git_line(READ_COMMIT, "internal/store/store.go", 2419)
    assert "groups" in line and "INTEGER PRIMARY KEY AUTOINCREMENT" in line, (
        f"internal/store/store.go:2419 no longer reads as the AUTOINCREMENT form: {line!r}"
    )


def test_mkfifo_flake_test_function_line():
    line = git_line(READ_COMMIT, "internal/interactive/replydrain_test.go", 86)
    assert line.strip().startswith(
        "func TestTerminalQueryInPaneOutputNeverStallsSession("
    ), f"replydrain_test.go:86 no longer names the flaking test function: {line!r}"


def test_mkfifo_flake_fatalf_line():
    line = git_line(READ_COMMIT, "internal/interactive/replydrain_test.go", 100)
    assert 't.Fatalf("Start: %v", err)' in line, (
        f"replydrain_test.go:100 no longer matches the flake's own log line: {line!r}"
    )


def test_gofmt_drift_files_are_untracked_spike_preview_only():
    for path in (
        ".spike-preview/cmd/conformance/main.go",
        ".spike-preview/conformance/conformance.go",
        ".spike-preview/conformance/conformance_test.go",
    ):
        proc = subprocess.run(
            ["git", "-C", REPO, "ls-files", "--error-unmatch", path],
            capture_output=True,
            text=True,
        )
        assert proc.returncode != 0, (
            f"{path} is unexpectedly tracked -- the gofmt-drift finding assumed untracked"
        )


def test_quantize_test_go_is_tracked_and_gofmt_clean_per_committed_log():
    proc = subprocess.run(
        ["git", "-C", REPO, "ls-files", "--error-unmatch", "internal/theme/quantize_test.go"],
        capture_output=True,
        text=True,
    )
    assert proc.returncode == 0, "internal/theme/quantize_test.go should be tracked"
    log_path = os.path.join(
        REPO, "docs/reports/phase4b-a2-findings/gofmt-quantize.log"
    )
    with open(log_path) as fh:
        content = fh.read()
    assert content.strip() == "EXIT:0", (
        f"gofmt-quantize.log no longer records a clean, exit-0 result: {content!r}"
    )
